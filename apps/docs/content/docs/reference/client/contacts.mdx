## `checkNumberStatus`

Needs an Insiders license. See [pricing](/docs/licensing/pricing).

<ReferenceCode>

```ts
client.checkNumberStatus(contactId: string): Promise<NumberCheck>
```

</ReferenceCode>

<h3 id="checknumberstatus-usage" className="scroll-m-28">Usage</h3>

<InterfaceTabs>
  <InterfaceTab value="SDK">

<ReferenceCode>

```ts
const result = await client.checkNumberStatus("447123456789@c.us");
```

</ReferenceCode>

  </InterfaceTab>
  <InterfaceTab value="SocketClient">

<ReferenceCode>

```ts
import { SocketClient } from '@open-wa/socket-client';

const client = await SocketClient.connect('http://localhost:8080', 'YOUR_API_KEY');
const result = await client.checkNumberStatus({
  contactId: "447123456789@c.us",
});
```

</ReferenceCode>

  </InterfaceTab>
  <InterfaceTab value="HTTP">

<ReferenceCode>

```bash
curl -G "http://localhost:8080/api/contacts/checkNumberStatus" \
  -H "x-api-key: YOUR_API_KEY" \
  --data-urlencode "contactId=447123456789@c.us"
```

</ReferenceCode>

Response:

<ReferenceCode>

```json
{
  "success": true,
  "data": {
    "id": {
      "server": "example",
      "user": "example",
      "_serialized": "example"
    },
    "canReceiveMessage": true
  }
}
```

</ReferenceCode>

  </InterfaceTab>
</InterfaceTabs>

| --- | --- | --- |
